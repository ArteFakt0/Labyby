function isPerfect($number) {
    if ($number <= 1) return false;
    $sum = 0;
    for ($i = 1; $i <= $number / 2; $i++) {
        if ($number % $i == 0) {
            $sum += $i;
        }
    }
    return $sum == $number;
}

$perfectNumbers = array_filter($array2, 'isPerfect');
echo "2. Досконалі числа у масиві: " . implode(', ', $perfectNumbers) . "\n";

// 9. Добуток елементів з парними індексами та вивід непарних індексів.
$array9 = [];
for ($i = 0; $i < 15; $i++) {
    $array9[] = rand(0, 100);
}

$evenIndexProduct = 1;
$hasPositiveEven = false;
$oddIndexPositives = [];

foreach ($array9 as $index => $value) {
    if ($value > 0) {
        if ($index % 2 === 0) {
            $evenIndexProduct *= $value;
            $hasPositiveEven = true;
        } else {
            $oddIndexPositives[] = $value;
        }
    }
}
if (!$hasPositiveEven) {
    $evenIndexProduct = 0; 
}

echo "9. Добуток (>0 з парними індексами): $evenIndexProduct\n";
echo "   Елементи (>0 з непарними індексами): " . implode(', ', $oddIndexPositives) . "\n";
